🛠️ Ready-to-Deploy Code Blueprints

This technical reference provides battle-tested production blueprints for implementing A2UI 1.0 across three primary environments: (1) React 19 with Tailwind tokens, (2) Flutter 3.x cross-platform client, and (3) Python Agent backend orchestrated via the Google Gen AI SDK v3.1.0 Interactions API.

Blueprint 1: React 19 Client Renderer (`@a2ui/react`)

The client application wraps its conversational interface in the A2UIProvider, registers custom components, and renders incoming agent surfaces natively:

// App.tsx — React 19 + TypeScript + @a2ui/react
import React from 'react';
import { A2UIProvider, A2UISurface, defineCatalog } from '@a2ui/react';
import { Button, Card, Badge, Switch } from '@/components/ui';

// 1. Define custom enterprise catalog mapping A2UI schema nodes to native React components
const enterpriseCatalog = defineCatalog({
  id: 'corp.enterprise.ds@1.0',
  components: {
    ApprovalCard: ({ title, amount, status, children }) => (
      <div className="rounded-2xl border border-sky-500/20 bg-slate-900/90 p-6 shadow-2xl backdrop-blur-xl">
        <div className="flex items-center justify-between border-b border-white/10 pb-4">
          <h3 className="text-lg font-semibold text-white">{title}</h3>
          <Badge variant={status === 'approved' ? 'success' : 'warning'}>{status}</Badge>
        </div>
        <div className="py-4 text-3xl font-bold tracking-tight text-sky-400">{amount}</div>
        <div className="space-y-4">{children}</div>
      </div>
    ),
    ActionButtons: ({ buttons, onAction }) => (
      <div className="flex justify-end gap-3 pt-4">
        {buttons.map((btn: any) => (
          <Button
            key={btn.action.functionName}
            variant={btn.variant}
            onClick={() => onAction(btn.action)}
          >
            {btn.label}
          </Button>
        ))}
      </div>
    )
  }
});

export function AgentChatPane({ a2uiSurfacePayload, onRpcDispatch }: any) {
  return (
    <A2UIProvider catalogs={[enterpriseCatalog]}>
      <div className="chat-container max-w-2xl mx-auto space-y-6">
        {/* Regular agent text messages here */}
        
        {/* Render interactive A2UI surface with bidirectional dispatch */}
        {a2uiSurfacePayload && (
          <A2UISurface
            payload={a2uiSurfacePayload}
            onAgentAction={async (action) => {
              // Dispatches typed RPC call back to the agent session
              await onRpcDispatch(action);
            }}
          />
        )}
      </div>
    </A2UIProvider>
  );
}

Blueprint 2: Python Agent Backend (`google-genai` v3.1.0)

Using the official Google Gen AI SDK v3.1.0, the backend agent prompts Gemini to emit an A2UI blueprint and handles bidirectional RPC callbacks:

# agent_backend.py — Python 3.12 + google-genai v3.1.0
import os
from google import genai
from google.genai import types

client = genai.Client(api_key=os.environ["GEMINI_API_KEY"])

A2UI_SCHEMA = {
    "type": "OBJECT",
    "properties": {
        "surfaceId": {"type": "STRING"},
        "catalogId": {"type": "STRING"},
        "layout": {"type": "OBJECT"}
    },
    "required": ["surfaceId", "catalogId", "layout"]
}

async def generate_interactive_approval(user_prompt: str):
    """Prompts Gemini 4 Argon to generate an A2UI 1.0 declarative surface."""
    response = client.interactions.create(
        model="gemini-4-argon",
        prompt=(
            f"User request: {user_prompt}\n"
            "Generate an A2UI 1.0 declarative surface blueprint using catalog 'corp.enterprise.ds@1.0'."
            "Include an ApprovalCard with title, amount, and an ActionButtons group triggering 'approveReservation'."
        ),
        response_mime_type="application/json",
        response_schema=A2UI_SCHEMA
    )
    return response.text

Blueprint 3: Model Context Protocol (MCP) App Configuration

A2UI 1.0 renderers are natively compatible with Model Context Protocol (MCP) Apps, allowing generative screens to render seamlessly inside Antigravity and Claude:

// mcp-app-manifest.json
{
  "name": "a2ui-enterprise-workbench",
  "version": "1.0.0",
  "protocol": "mcp/1.0",
  "capabilities": {
    "generativeUI": {
      "supportedProtocols": ["a2ui/1.0"],
      "supportedCatalogs": [
        "corp.enterprise.ds@1.0",
        "corp.finance.approval@1.0",
        "@a2ui/core@1.0"
      ],
      "compressionFormats": ["a2ui-express/1.0"]
    }
  }
}
