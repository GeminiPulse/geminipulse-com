Blueprint 2: Fleet Route Optimizer with Routes API & Shift Constraints

Demonstrates querying the Google Routes API (v2) from an autonomous Python agent (`google-genai` v3.0.0) with waypoint optimization and traffic awareness:

from google import genai
from google.genai import types
import requests

def optimize_fleet_route(origin_lat, origin_lng, destination_lat, destination_lng, waypoints, gcp_maps_key):
    """
    Computes a multi-stop optimal delivery route respecting shift windows.
    """
    url = "https://routes.googleapis.com/directions/v2:computeRoutes"
    headers = {
        "Content-Type": "application/json",
        "X-Goog-Api-Key": gcp_maps_key,
        "X-Goog-FieldMask": "routes.duration,routes.distanceMeters,routes.polyline.encodedPolyline,routes.optimizedIntermediateWaypointIndex"
    }
    payload = {
        "origin": {"location": {"latLng": {"latitude": origin_lat, "longitude": origin_lng}}},
        "destination": {"location": {"latLng": {"latitude": destination_lat, "longitude": destination_lng}}},
        "intermediates": [
            {"location": {"latLng": {"latitude": wp["lat"], "longitude": wp["lng"]}}, "via": False}
            for wp in waypoints
        ],
        "travelMode": "DRIVE",
        "routingPreference": "TRAFFIC_AWARE_OPTIMAL",
        "optimizeWaypointOrder": True
    }
    response = requests.post(url, headers=headers, json=payload)
    return response.json()

# Calling with Gemini 4 Argon via Gen AI Python SDK v3.0.0
client = genai.Client()
response = client.models.generate_content(
    model="gemini-4-argon",
    contents="Generate an optimal 4-stop dispatch route starting from Brooklyn depot to Queens delivery points.",
    config=types.GenerateContentConfig(
        tools=[optimize_fleet_route],
        temperature=0.2
    )
)

Blueprint 3: Vector Store Locator with Places API (New)

Uses the modernized Places API (New) for conversational local discovery and operational status filtering:

import { Client } from '@googlemaps/google-maps-services-js';

const client = new Client({});

async function findOpenStoresNearby(lat: number, lng: number, radiusMeters: number, apiKey: string) {
  const response = await fetch('https://places.googleapis.com/v1/places:searchNearby', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'X-Goog-Api-Key': apiKey,
      'X-Goog-FieldMask': 'places.displayName,places.formattedAddress,places.currentOpeningHours,places.rating'
    },
    body: JSON.stringify({
      includedTypes: ['convenience_store', 'supermarket'],
      maxResultCount: 10,
      locationRestriction: {
        circle: {
          center: { latitude: lat, longitude: lng },
          radius: radiusMeters
        }
      }
    })
  });
  return await response.json();
}
