Use Case 1: Symmetrical PR Verification Gate (@google/genai v3.6.1)

Adopt Sashiko’s core design inside your CI pipeline: separate fast single-focus discovery passes (requiring each pass to report both active concerns and dismissed candidates) from an independent gemini-4-carbon triage pass that audits shaky self-dismissals.

import { GoogleGenAI } from "@google/genai";

const ai = new GoogleGenAI({});
const FOCUSES = ["locking_and_races", "resource_lifetimes", "security_boundaries"];

export async function runSashikoPatternReview(diffBundle, subsystemGuide) {
  const specialists = await Promise.all(
    FOCUSES.map((focus) =>
      ai.interactions.create({
        model: "gemini-4-hydrogen",
        input: JSON.stringify({
          focus,
          subsystemGuide,
          diffBundle,
          instruction: "Return concerns[] and dismissed_candidates[] citing exact callers.",
        }),
        generation_config: { thinking_level: "medium", max_output_tokens: 16384 },
      })
    )
  );

  return ai.interactions.create({
    model: "gemini-4-carbon",
    input: JSON.stringify({
      stage: "symmetrical_triage",
      rule: "Promote any dismissal that assumes an unverified caller invariant.",
      reports: specialists.map((s, idx) => ({ focus: FOCUSES[idx], output: s.output_text })),
    }),
    generation_config: { thinking_level: "high", max_output_tokens: 32768 },
  });
}

Use Case 2: Bounded Supervisory Loop in Google Antigravity SDK (Python)

Follow Alex Astrum’s verified harness split: route read-only codebase exploration to gemini-4-hydrogen, feature changes to gemini-4-carbon, enforce a deterministic outside test gate with max_retries=1, and cap total spend via BudgetConfig.

import asyncio
import subprocess
from google.antigravity import Agent, LocalAgentConfig, types
from google.antigravity.hooks import policy

def verify_output(workspace_path: str) -> tuple[bool, str]:
    result = subprocess.run(
        ["go", "test", "-race", "."], cwd=workspace_path, capture_output=True, text=True
    )
    return result.returncode == 0, (result.stdout + "\n" + result.stderr).strip()

researcher = types.SubagentConfig(
    name="researcher",
    description="Searches the codebase and summarizes implementation paths.",
    system_instructions="Analyze call sites and report back without editing files.",
    model="gemini-4-hydrogen",
    capabilities=types.SubagentCapabilities(
        enabled_tools=[types.BuiltinTools.VIEW_FILE, types.BuiltinTools.SEARCH_DIR, types.BuiltinTools.FIND_FILE],
    ),
)

feature_implementer = types.SubagentConfig(
    name="feature_implementer",
    description="Implements scoped edits while keeping all tests passing.",
    system_instructions="Follow AGENTS.md, fix failures cleanly, and never weaken tests.",
    model="gemini-4-carbon",
    capabilities=types.SubagentCapabilities(
        enabled_tools=[
            types.BuiltinTools.VIEW_FILE,
            types.BuiltinTools.EDIT_FILE,
            types.BuiltinTools.CREATE_FILE,
            types.BuiltinTools.RUN_COMMAND,
        ],
    ),
)

supervisor_config = LocalAgentConfig(
    model="gemini-4-carbon",
    workspaces=["./workspace"],
    subagents=[researcher, feature_implementer],
    policies=[policy.allow_all()],
    budget_config=types.BudgetConfig(max_model_calls=20, max_total_tokens=200_000),
    system_instructions="Delegate search to @researcher and edits to @feature_implementer.",

Use Case 3: Envoy 2026-07-28 MCP Header Dispatch & HeaderMismatch Guard

Attach Mcp-Method and Mcp-Name headers so Envoy routes MCP traffic at line rate without buffering large JSON-RPC payloads, while verifying header-body equality on the backend to reject tampering with HTTP 400 / -32020 (HeaderMismatch).

export function verifyMcpHeaderParity(headers, rpcBody) {
  const hdrMethod = headers.get("Mcp-Method");
  const hdrName = headers.get("Mcp-Name");
  const bodyName = rpcBody?.params?.name || "";

  if (hdrMethod !== rpcBody?.method || (hdrName && hdrName !== bodyName)) {
    return new Response(
      JSON.stringify({
        jsonrpc: "2.0",
        id: rpcBody?.id ?? null,
        error: { code: -32020, message: "HeaderMismatch: Mcp-Method/Mcp-Name mismatch" },
      }),
      { status: 400, headers: { "Content-Type": "application/json" } }
    );
  }
  return null;
}
