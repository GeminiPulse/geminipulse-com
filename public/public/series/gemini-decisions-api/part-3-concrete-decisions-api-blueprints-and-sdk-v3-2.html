Blueprint 1: Multimodal Safety Inspection in Python (v3.2.0)

This blueprint demonstrates evaluating a shared multimodal input (JPEG image + accompanying inspection text note) against all three decision primitives (noul, choice, and score) in a single request:

# pip install google-genai>=3.2.0
from google import genai

client = genai.Client(http_options={"api_version": "v1beta"})

result = client.decisions.create(
    model="gemini-decision-1",
    input=[
        {
            "type": "image",
            "uri": "https://storage.googleapis.com/generativeai-downloads/images/spill.jpg",
            "mime_type": "image/jpeg",
        },
        {
            "type": "text",
            "text": "Breakroom counter check before the 9 AM executive meeting.",
        },
    ],
    questions={
        "has_spill": {
            "type": "noul",
            "instructions": "Is there an uncleaned liquid spill on the counter?",
            "criteria": {
                "true": "Visible puddle, liquid rings, or active stain on surface.",
                "false": "Surface is dry, wiped clean, or holds only closed containers."
            }
        },
        "next_action": {
            "type": "choice",
            "instructions": "Which cleanup action best fits the scene?",
            "criteria": {
                "wipe_with_cloth": "Small counter spill with a cloth or paper towel already nearby.",
                "dispatch_janitorial": "Large floor hazard, broken glass, chemical hazard, or active ceiling leak.",
                "ready_for_meeting": "Counter is clean and ready for arriving guests."
            }
        },
        "severity": {
            "type": "score",
            "instructions": "Rate the severity of the counter mess from 0 to 2.",
            "criteria": [
                "Clean and ready.",
                "Minor localized spill requiring quick wipe.",
                "Major spill or physical safety hazard."
            ]
        }
    }
)

# Inspect calibrated answers with zero output token fees
has_spill = result.answers["has_spill"]
print(f"Spill Present: {has_spill.noul > 0.5} (Confidence: {has_spill.confidence:.2f})")

action = result.answers["next_action"]
print(f"Recommended Action: {action.winner} (Probability: {action.probabilities[action.winner]:.2f})")

severity = result.answers["severity"]
print(f"Severity Score: {severity.score:.2f} / 2.0 (Confidence: {severity.confidence:.2f})")

Blueprint 2: High-Speed Model Router in TypeScript (@google/genai v3.2.0)

Replicate the 40% latency reduction achieved by Gemini Notebook by routing incoming queries to the optimal model and thinking depth:

// npm install @google/genai@3.2.0
import { GoogleGenAI } from '@google/genai';

const ai = new GoogleGenAI({
  apiKey: process.env.GEMINI_API_KEY,
  httpOptions: { apiVersion: 'v1beta' }
});

export async function routeUserPrompt(userPrompt: string) {
  const decision = await ai.decisions.create({
    model: 'gemini-decision-1',
    input: userPrompt,
    questions: {
      target_tier: {
        type: 'choice',
        instructions: 'Determine the optimal model tier for this user request based on reasoning complexity.',
        criteria: {
          flash_lite: 'Simple factual queries, greeting, short translations, trivial string formatting.',
          flash: 'Standard multi-paragraph summarization, email drafting, conversational chat.',
          carbon_low_think: 'Standard code generation, unit test writing, SQL query synthesis.',
          carbon_high_think: 'Deep mathematical proofs, complex system architecture, formal verification, multi-file refactoring.'
        }
      },
      requires_grounding: {
        type: 'noul',
        instructions: 'Does this query ask about recent events, live stock prices, current sports scores, or specific web facts?'
      }
    }
  });

  const tier = decision.answers?.target_tier?.winner;
  const needSearch = (decision.answers?.requires_grounding?.noul ?? 0) > 0.75;

  console.log(`Routed to: ${tier} | Google Search Grounding: ${needSearch}`);
  return { tier, needSearch };
}

Blueprint 3: Self-Hosted BYOS Environments in SDK v3.2.0

For organizations operating under strict data sovereignty, HIPAA compliance, or proprietary hardware constraints, SDK v3.2.0 introduces self-hosted BYOS environments. Instead of executing code inside Google-hosted on-demand microVMs, your agent connects to your private on-prem or VPC compute cluster:

import { GoogleGenAI } from '@google/genai';

const ai = new GoogleGenAI({ apiKey: process.env.GEMINI_API_KEY });

// Create a Self-Hosted BYOS Environment
const selfHostedEnv = await ai.environments.create({
  mode: 'self_hosted',
  system_instruction: 'Execute Python analysis scripts strictly inside the private VPC sandbox with restricted egress.',
  custom_config: {
    cluster_id: 'prod-vpc-us-east-4',
    runner_group: 'airgapped-cuda-nodes',
    max_memory_mb: 32768,
    egress_policy: 'internal_only'
  }
});

console.log(`BYOS Environment provisioned: ${selfHostedEnv.id} (Status: ${selfHostedEnv.status})`);

// Bind private environment to an autonomous agent interaction
const interaction = await ai.interactions.create({
  model: 'gemini-4-carbon',
  environment: selfHostedEnv.id,
  input: 'Analyze the private enterprise sales ledger in ledger.parquet and generate executive financial metrics.'
});
