kind: WorkerPool
metadata:
  name: general-coding-pool
  namespace: substrate-workloads
spec:
  replicas: 20
  slotsPerWorker: 10
  runtimeClassName: gvisor
  template:
    spec:
      containers:
      - name: sandbox-runtime
        image: gcr.io/gemini-pulse-prod/agent-runtime:v2.0
        resources:
          requests:
            cpu: "2"
            memory: "8Gi"
          limits:
            cpu: "4"
            memory: "16Gi"
---

  name: workspace-antigravity-dev-442
  namespace: substrate-workloads

  targetWorkerPool: general-coding-pool
  idleTimeoutSeconds: 30
