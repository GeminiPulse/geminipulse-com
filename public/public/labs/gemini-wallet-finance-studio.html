TypeScript implementation using @google/genai v3.3.0 and the Interactions API with read-only wallet tools: 
import { GoogleGenAI } from "@google/genai";

const client = new GoogleGenAI({ apiKey: process.env.GEMINI_API_KEY });

// Read-only tool definitions
const walletTools = [{
  name: "fetchTransactions",
  description: "Read-only ledger from linked Google Wallet cards.",
  parameters: {
    type: "OBJECT",
    properties: {
      category: { type: "STRING" },
      lookbackDays: { type: "INTEGER" }
    }
  }
}];

const interaction = await client.interactions.create({
  model: "gemini-4-hydrogen",
  input: "@Google Wallet Review spending to save $1k.",
  config: {
    system_instruction: "Strict read-only. Never mutate accounts.",
    thinking_level: "high",
    automatic_continuation: true,
    tools: walletTools
  }
});
